# run-log 实施计划 — Part A（记录与包裹执行）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建成 `run-log` 的记录与包裹执行：把一次计算运行变成一条**耐久**的记录（命令、配置、种子、git 状态、退出码、时长），并在命令启动**之前**就把它落盘。

**Architecture:** 一运行一文件（`experiments/log/<run_id>.json`），用 `open(path, "x")` 做原子占位；命令启动前写 `status: "running"`，结束后原子更新为 `completed`/`failed`。配置以"路径 + sha256 + 解析后的内容"三件套记录，使记录自包含。git 同时记 commit 与**脏标志**。

**Tech Stack:** Python 3.12（`tools/.venv`）；stdlib `subprocess` / `json` / `hashlib` / `os` / `argparse`；unittest。

```python
import json
import tempfile
import unittest
from pathlib import Path

from ccfa.run_log import git_state, new_run_id, read_config, run_command


class BaseCase(unittest.TestCase):
    def setUp(self):
        self.root = Path(tempfile.mkdtemp())
        self.log_dir = self.root / "experiments" / "log"
        self.config = self.root / "configs" / "a.yaml"
        self.config.parent.mkdir(parents=True, exist_ok=True)
        self.config.write_text("epochs: 5\n", encoding="utf-8")

    def runner(self, exit_code=0, output="ok\n"):
        calls = []

        def run(argv, cwd):
            calls.append(list(argv))
            return exit_code, output

        return run, calls


class TestRunId(BaseCase):
    def test_first_id_uses_stamp_and_counter(self):
        run_id, path = new_run_id(self.log_dir, stamp="20261003T142530")
        self.assertEqual(run_id, "20261003T142530-01")
        self.assertTrue(path.is_file())  # the claim file exists immediately
        self.assertEqual(json.loads(path.read_text(encoding="utf-8")), {})

    def test_second_id_does_not_clobber_the_first(self):
        first_id, first_path = new_run_id(self.log_dir, stamp="20261003T142530")
        second_id, second_path = new_run_id(self.log_dir, stamp="20261003T142530")
        self.assertNotEqual(first_id, second_id)
        self.assertEqual(second_id, "20261003T142530-02")
        self.assertEqual(json.loads(first_path.read_text(encoding="utf-8")), {})


class TestGitState(BaseCase):
    def test_non_repo_returns_none_and_false(self):
        commit, dirty = git_state(self.root)
        self.assertIsNone(commit)
        self.assertFalse(dirty)


class TestReadConfig(BaseCase):
    def test_records_path_hash_and_parsed_content(self):
        entry = read_config(self.config, paper_root=self.root)
        self.assertEqual(entry["path"], "configs/a.yaml")
        self.assertEqual(entry["content"], {"epochs": 5})
        self.assertTrue(entry["sha256"].startswith("sha256:"))

    def test_missing_config_is_a_tool_error(self):
        with self.assertRaises(ValueError):
            read_config(self.root / "nope.yaml", paper_root=self.root)


class TestRunCommand(BaseCase):
    def _record(self, run_id):
        return json.loads((self.log_dir / f"{run_id}.json").read_text(encoding="utf-8"))

    def test_records_a_completed_run(self):
        run, calls = self.runner(exit_code=0)
        run_id, record = run_command(
            ["python", "train.py"],
            self.log_dir,
            self.root,
            config=self.config,
            seed="42",
            runner=run,
        )
        self.assertEqual(calls, [["python", "train.py"]])
        self.assertEqual(record["status"], "completed")
        self.assertEqual(record["exit_code"], 0)
        self.assertEqual(record["seed"], "42")
        self.assertIsNone(record["metrics"])
        self.assertIsNotNone(record["finished_at"])
        self.assertEqual(self._record(run_id)["run_id"], run_id)

    def test_records_a_failed_run_without_raising(self):
        run, _ = self.runner(exit_code=3)
        _, record = run_command(["false"], self.log_dir, self.root, runner=run)
        self.assertEqual(record["status"], "failed")
        self.assertEqual(record["exit_code"], 3)

    def test_running_record_exists_before_the_command_finishes(self):
        seen = {}

        def run(argv, cwd):
            # At the moment the command starts, the record must already say running.
            paths = list(self.log_dir.glob("*.json"))
            seen["count"] = len(paths)
            seen["status"] = json.loads(paths[0].read_text(encoding="utf-8"))["status"]
            return 0, ""

        run_command(["python", "train.py"], self.log_dir, self.root, runner=run)
        self.assertEqual(seen["count"], 1)
        self.assertEqual(seen["status"], "running")

    def test_command_is_recorded_relative_to_paper_root(self):
        run, _ = self.runner()
        _, record = run_command(["python", "train.py"], self.log_dir, self.root, runner=run)
        self.assertEqual(record["cwd"], ".")

    def test_unwritable_log_dir_is_a_tool_error(self):
        # A file used as a directory parent fails deterministically on every OS,
        # unlike a root-level path whose permissions vary by machine.
        blocker = self.root / "blocker"
        blocker.write_text("not a directory\n", encoding="utf-8")
        run, _ = self.runner()
        with self.assertRaises(ValueError):
            run_command(["python", "train.py"], blocker / "log", self.root, runner=run)


if __name__ == "__main__":
    unittest.main()
```

先在 `tools/ccfa/cli.py` 末尾加共享的原子写（`provenance.py` 里已有一份同语义的实现，改为调用它）：

在 `cli.py` 顶部 import 区加 `import os` 与 `from pathlib import Path`（当前只有 `json` / `sys` / `typing`），然后在文件末尾加：

```python
def save_text_atomically(path: Path, text: str, description: str = "文件") -> None:
    """Write *text* to *path* atomically, folding OSError into ValueError."""
    path = Path(path)
    try:
        path.parent.mkdir(parents=True, exist_ok=True)
        temporary = path.with_name(path.name + ".tmp")
        temporary.write_text(text, encoding="utf-8")
        os.replace(temporary, path)
    except OSError as exc:
        raise ValueError(f"无法写入{description} {path}: {exc}") from exc
```

`provenance.py` 里那份同语义实现改为调用它；**行为必须逐位不变**（同一临时文件名、同一 `mkdir`、同一错误消息：注意 `provenance` 原来的描述词是 `"store"`，调用时要显式传 `description="store"` 才能保住原句）。评审者会专门核对这一点。

再创建 `tools/ccfa/run_log.py`，实现上列六个接口。要点：

- `new_run_id`：`log_dir.mkdir(parents=True, exist_ok=True)`；循环 `-01`、`-02`…，用 `open(path, "x")` 占位并写入 `{}`；全部被占则继续递增。
- `git_state`：`subprocess.run(["git", "-C", str(paper_root), "rev-parse", "HEAD"], ...)` 与 `git status --porcelain`，两者都要吞掉 `OSError`（机器上没装 git）与超时，失败返回 `(None, False)`。
- `read_config`：读文件 → `sha256_of` 同款哈希 → `yaml.safe_load`；路径相对 `paper_root`，绝对路径或读不到都抛 `ValueError`。
- `run_command`：先 `new_run_id` → 组装 `running` 记录 → `cli.save_text_atomically` 落盘 → 调用 `runner`（默认真实 `subprocess.run`，`cwd=paper_root`）→ 计算时长 → 更新 `finished_at`/`duration_s`/`exit_code`/`status` → 再次原子落盘 → 返回 `(run_id, record)`。**不得**因被包裹命令非零而抛异常。
- `main`：`--log-dir`（必需）、`--paper-root`（必需）为顶层参数；子命令 `run`，其 `--config` / `--seed` / `--notes` / `--propagate-exit` 后跟 `--` 与命令（用 `nargs=argparse.REMAINDER`）。默认退 0，`--propagate-exit` 时退被包裹命令的码，工具自身失败退 2（`tool_error`）。

## 验收（Part A）

期望：基线 **342** + 本任务新增 **11**，共 **353** 个测试通过。

- **耐久性**：命令启动前记录已在盘上且 `status == "running"`——进程被杀、断电、卡死都留痕（§9.2）。判别力检查会实测这一点。
- **自包含**：`config` 三件套（路径 + 哈希 + 解析后内容）让记录不依赖配置文件仍在原处。
- **脏树可见**：`git_dirty` 进记录，复现时能看出"当时的树不是干净的"。
- **不覆盖**：run_id 用 `O_EXCL` 占位，两个进程不会认领同一个 id。
- **退出码例外**（E11）：默认退 0（记录成功即成功），`--propagate-exit` 才透传被包裹命令的码；工具自身失败退 2。

已知边界（留给最终全分支评审）：`config` 只支持 YAML（要支持 JSON/TOML 需另加分支）；`git_dirty` 只看有无改动，不记录改了什么；记录里的 `command` 原样保存 argv，不解析、不校验可执行文件是否存在（那是 `runner` 的职责）。

**一条必须传给 Part B 的边界**：`new_run_id` 用 `O_EXCL` 建出的占位文件内容先是 `{}`；若进程在"占位"与"写入 running 记录"之间被杀，盘上会留下一个**没有 `status` 字段**的记录。Part B 的 `check` 必须把这种记录报成 problem（建议 code `run-log-malformed`），否则这类损坏会静默通过——而静默通过正是这一层存在的理由所要防的事。
